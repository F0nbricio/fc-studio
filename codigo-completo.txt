CREATE TABLE IF NOT EXISTS project_briefings (
  id TEXT PRIMARY KEY,
  client_name TEXT NOT NULL CHECK (length(trim(client_name)) >= 2),
  client_whatsapp TEXT NOT NULL CHECK (length(trim(client_whatsapp)) >= 8),
  client_email TEXT,
  project_type TEXT NOT NULL,
  business_goal TEXT NOT NULL CHECK (length(trim(business_goal)) >= 5),
  desired_features TEXT NOT NULL DEFAULT '[]',
  design_style TEXT NOT NULL DEFAULT 'Moderno e Minimalista',
  has_domain_or_brand INTEGER NOT NULL DEFAULT 0 CHECK (has_domain_or_brand IN (0, 1)),
  additional_notes TEXT,
  generated_gemini_prompt TEXT NOT NULL CHECK (length(trim(generated_gemini_prompt)) >= 20),
  status TEXT NOT NULL DEFAULT 'novo' CHECK (status IN ('novo', 'em_analise', 'proposta_enviada', 'feito', 'em_producao', 'fechado')),
  agreed_amount_brl REAL CHECK (agreed_amount_brl IS NULL OR agreed_amount_brl >= 0),
  commercial_notes TEXT,
  created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))
);

CREATE INDEX IF NOT EXISTS idx_project_briefings_status_created
  ON project_briefings (status, created_at DESC);

import crypto from 'crypto';
import { NextResponse } from 'next/server';
import { BriefingFormData } from '@/types/briefing';
import { buildGeminiProjectPrompt, buildWhatsappShareLink } from '@/lib/prompt-builder';
import { checkRateLimit, extractClientIp } from '@/lib/security/rate-limit';
import {
  createPublicBriefing,
  isValidBriefingStatus,
  listAdminBriefings,
  updateBriefingCommercialDetails,
} from '@/lib/turso/repository';
import { extractPhoneDigits } from '@/lib/phone-mask';

function isAuthorizedAdmin(request: Request): boolean {
  const expectedSecret = process.env.ADMIN_API_SECRET?.trim();
  const providedSecret = request.headers.get('x-admin-token')?.trim();

  if (!expectedSecret || !providedSecret) return false;
  if (expectedSecret.length !== providedSecret.length) return false;

  return crypto.timingSafeEqual(
    Buffer.from(expectedSecret, 'utf8'),
    Buffer.from(providedSecret, 'utf8')
  );
}

function enforceAdminRateLimit(request: Request, action: string) {
  const ip = extractClientIp(request);
  const check = checkRateLimit(action + ":" + ip, 30, 60_000);
  if (!check.allowed) {
    return NextResponse.json(
      {
        success: false,
        message: 'Muitas tentativas em sequência. Aguarde ' + check.retryAfterSec + 's e tente novamente.',
      },
      {
        status: 429,
        headers: { 'Retry-After': String(check.retryAfterSec), 'Cache-Control': 'no-store' },
      }
    );
  }
  return null;
}

export async function GET(request: Request) {
  const limited = enforceAdminRateLimit(request, 'admin-get');
  if (limited) return limited;

  if (!isAuthorizedAdmin(request)) {
    return NextResponse.json(
      { success: false, message: 'Senha incorreta ou não informada.' },
      { status: 401, headers: { 'Cache-Control': 'no-store' } }
    );
  }

  try {
    const { searchParams } = new URL(request.url);
    const limit = Number(searchParams.get('limit') || '50');
    const briefings = await listAdminBriefings(limit);

    return NextResponse.json(
      {
        success: true,
        total: briefings.length,
        briefings,
      },
      { headers: { 'Cache-Control': 'no-store' } }
    );
  } catch {
    return NextResponse.json(
      { success: false, message: 'Erro ao consultar briefings no banco.' },
      { status: 500, headers: { 'Cache-Control': 'no-store' } }
    );
  }
}

export async function PATCH(request: Request) {
  const limited = enforceAdminRateLimit(request, 'admin-patch');
  if (limited) return limited;

  if (!isAuthorizedAdmin(request)) {
    return NextResponse.json(
      { success: false, message: 'Acesso não autorizado.' },
      { status: 401, headers: { 'Cache-Control': 'no-store' } }
    );
  }

  try {
    const body = (await request.json()) as {
      id?: string;
      status?: string;
      agreedAmountBrl?: number | string | null;
      commercialNotes?: string | null;
    };

    const id = (body.id || '').trim();
    const status = (body.status || '').trim();

    if (!id || !isValidBriefingStatus(status)) {
      return NextResponse.json(
        { success: false, message: 'ID ou status inválido.' },
        { status: 400, headers: { 'Cache-Control': 'no-store' } }
      );
    }

    let agreedAmountBrl: number | null = null;
    if (
      body.agreedAmountBrl !== undefined &&
      body.agreedAmountBrl !== null &&
      String(body.agreedAmountBrl).trim() !== ''
    ) {
      const normalizedNumber = Number(String(body.agreedAmountBrl).replace(',', '.'));
      if (!Number.isFinite(normalizedNumber) || normalizedNumber < 0) {
        return NextResponse.json(
          { success: false, message: 'Informe um valor válido em reais.' },
          { status: 400, headers: { 'Cache-Control': 'no-store' } }
        );
      }
      agreedAmountBrl = Math.round(normalizedNumber * 100) / 100;
    }

    const sanitizedNotes =
      typeof body.commercialNotes === 'string' && body.commercialNotes.trim().length > 0
        ? body.commercialNotes.replace(/[<>]/g, '').trim().slice(0, 1000)
        : null;

    const updated = await updateBriefingCommercialDetails(id, {
      status,
      agreedAmountBrl,
      commercialNotes: sanitizedNotes,
    });

    if (!updated.ok) {
      return NextResponse.json(
        { success: false, message: 'Não foi possível atualizar os dados no banco.' },
        { status: updated.reason === 'NOT_FOUND' ? 404 : 500, headers: { 'Cache-Control': 'no-store' } }
      );
    }

    return NextResponse.json(
      {
        success: true,
        id,
        status,
        agreedAmountBrl,
        commercialNotes: sanitizedNotes,
        message: 'Acompanhamento comercial salvo com sucesso.',
      },
      { headers: { 'Cache-Control': 'no-store' } }
    );
  } catch {
    return NextResponse.json(
      { success: false, message: 'Erro ao processar atualização.' },
      { status: 500, headers: { 'Cache-Control': 'no-store' } }
    );
  }
}

export async function POST(request: Request) {
  try {
    const body = (await request.json()) as Partial<BriefingFormData>;

    const clientName = (body.clientName || '').trim();
    const rawWhatsapp = (body.clientWhatsapp || '').trim();
    const phoneDigits = extractPhoneDigits(rawWhatsapp);
    const projectType = (body.projectType || '').trim();
    const businessGoal = (body.businessGoal || '').trim();

    if (clientName.length < 2) {
      return NextResponse.json(
        { success: false, message: 'Informe seu nome para continuarmos.' },
        { status: 400 }
      );
    }

    if (phoneDigits.length < 10) {
      return NextResponse.json(
        {
          success: false,
          message: 'Informe um número de celular válido com DDD (10 ou 11 dígitos).',
        },
        { status: 400 }
      );
    }

    if (!projectType || businessGoal.length < 5) {
      return NextResponse.json(
        {
          success: false,
          message: 'Descreva o tipo de site e o objetivo principal.',
        },
        { status: 400 }
      );
    }

    const sanitizedData: BriefingFormData = {
      clientName: clientName.slice(0, 120),
      clientWhatsapp: phoneDigits,
      clientEmail: body.clientEmail ? body.clientEmail.trim().slice(0, 150) : undefined,
      projectType: projectType.slice(0, 100),
      businessGoal: businessGoal.slice(0, 1000),
      desiredFeatures: Array.isArray(body.desiredFeatures)
        ? body.desiredFeatures.slice(0, 15).map((item) => String(item).trim().slice(0, 120))
        : [],
      designStyle: (body.designStyle || 'Moderno e Minimalista').trim().slice(0, 80),
      hasDomainOrBrand: Boolean(body.hasDomainOrBrand),
      additionalNotes: body.additionalNotes
        ? body.additionalNotes.trim().slice(0, 1500)
        : undefined,
    };

    const privatePrompt = buildGeminiProjectPrompt(sanitizedData);
    const ownerWhatsapp = process.env.NEXT_PUBLIC_OWNER_WHATSAPP || '';
    const whatsappUrl = buildWhatsappShareLink(ownerWhatsapp, sanitizedData);

    const dbResult = await createPublicBriefing(sanitizedData, privatePrompt);

    return NextResponse.json({
      success: true,
      persisted: dbResult.ok,
      whatsappUrl,
      message: 'Solicitação enviada! Redirecionando para o WhatsApp...',
    });
  } catch {
    return NextResponse.json(
      { success: false, message: 'Não foi possível processar sua solicitação agora.' },
      { status: 500 }
    );
  }
}

interface AdminBriefingItem {
  id: string;
  client_name: string;
  client_whatsapp: string;
  project_type: string;
  business_goal: string;
  desired_features: string[];
  design_style: string;
  has_domain_or_brand: boolean;
  additional_notes: string | null;
  generated_gemini_prompt: string;
  status: BriefingStatus;
  agreed_amount_brl: number | null;
  commercial_notes: string | null;
  created_at: string;
}

interface DraftCommercialState {
  status: BriefingStatus;
  amountInput: string;
  notesInput: string;
}

type FilterMode = 'todos' | BriefingStatus;

function buildDraftsMap(items: AdminBriefingItem[]): Record<string, DraftCommercialState> {
  const map: Record<string, DraftCommercialState> = {};
  for (const item of items) {
    map[item.id] = {
      status: item.status,
      amountInput:
        item.agreed_amount_brl !== null && item.agreed_amount_brl !== undefined
          ? String(item.agreed_amount_brl)
          : '',
      notesInput: item.commercial_notes || '',
    };
  }
  return map;
}

export function AdminBriefingsPanel() {
  const [password, setPassword] = useState('');
  const [authenticated, setAuthenticated] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [feedback, setFeedback] = useState<string | null>(null);
  const [briefings, setBriefings] = useState<AdminBriefingItem[]>([]);
  const [drafts, setDrafts] = useState<Record<string, DraftCommercialState>>({});
  const [filterMode, setFilterMode] = useState<FilterMode>('todos');
  const [copiedPromptId, setCopiedPromptId] = useState<string | null>(null);
  const [copiedProposalId, setCopiedProposalId] = useState<string | null>(null);
  const [savingId, setSavingId] = useState<string | null>(null);

  async function fetchBriefings(secret: string) {
    setLoading(true);
    setError(null);
    setFeedback(null);

    try {
      const response = await fetch('/api/briefings?limit=50', {
        method: 'GET',
        headers: {
          'x-admin-token': secret.trim(),
        },
        cache: 'no-store',
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        setError(data.message || 'Senha incorreta.');
        setAuthenticated(false);
        setLoading(false);
        return;
      }

      const list: AdminBriefingItem[] = Array.isArray(data.briefings) ? data.briefings : [];
      setBriefings(list);
      setDrafts(buildDraftsMap(list));
      setAuthenticated(true);
    } catch {
      setError('Erro de conexão ao tentar entrar.');
    } finally {
      setLoading(false);
    }
  }

  async function handleLogin(e: React.FormEvent) {
    e.preventDefault();
    if (!password.trim()) {
      setError('Digite sua senha para continuar.');
      return;
    }
    await fetchBriefings(password);
  }

  async function handleCopyPrompt(item: AdminBriefingItem) {
    await navigator.clipboard.writeText(item.generated_gemini_prompt);
    setCopiedPromptId(item.id);
    setTimeout(() => setCopiedPromptId(null), 2500);
  }

  async function handleCopyProposal(itemId: string, proposalText: string) {
    await navigator.clipboard.writeText(proposalText);
    setCopiedProposalId(itemId);
    setTimeout(() => setCopiedProposalId(null), 2500);
  }

  function updateDraft(
    itemId: string,
    patch: Partial<DraftCommercialState>,
    fallbackItem: AdminBriefingItem

    setDrafts((prev) => {
      const current = prev[itemId] || {
        status: fallbackItem.status,
        amountInput:
          fallbackItem.agreed_amount_brl !== null && fallbackItem.agreed_amount_brl !== undefined
            ? String(fallbackItem.agreed_amount_brl)
            : '',
        notesInput: fallbackItem.commercial_notes || '',
      };

      return {
        ...prev,
        [itemId]: {
          ...current,
          ...patch,
        },
      };
    });
  }

  async function handleSaveCommercial(item: AdminBriefingItem) {
    const draft = drafts[item.id] || {
      status: item.status,
      amountInput: item.agreed_amount_brl !== null ? String(item.agreed_amount_brl) : '',
      notesInput: item.commercial_notes || '',
    };

    setSavingId(item.id);
    setError(null);
    setFeedback(null);

    try {
      const response = await fetch('/api/briefings', {
        method: 'PATCH',
        headers: {
          'Content-Type': 'application/json',
          'x-admin-token': password.trim(),
        },
        body: JSON.stringify({
          id: item.id,
          status: draft.status,
          agreedAmountBrl: draft.amountInput.trim() === '' ? null : draft.amountInput.trim(),
          commercialNotes: draft.notesInput.trim() === '' ? null : draft.notesInput.trim(),

'use client';

import React, { useState } from 'react';
import { BriefingFormData, BriefingSubmissionResult } from '@/types/briefing';
import { extractPhoneDigits, maskPhoneBr } from '@/lib/phone-mask';

const PROJECT_TYPES = [
  'Site Institucional / Empresa',
  'Landing Page de Alta Conversão',
  'Loja Virtual / E-commerce',
  'Portfólio Profissional',
  'Sistema Web / Agendamento / Catálogo',
  'Outro Projeto Sob Demanda',
];

const FEATURE_OPTIONS = [
  'Botão direto para WhatsApp',
  'Formulário de orçamento / contato',
  'Catálogo de produtos ou serviços',
  'Integração com pagamentos (Pix / Cartão)',
  'Área administrativa para editar conteúdos',
  'Otimização para aparecer no Google (SEO)',
  'Mapa de localização e redes sociais',
  'Depoimentos e provas sociais',
];

const DESIGN_STYLES = [
  'Moderno e Minimalista',
  'Corporativo e Sério',
  'Criativo e Vibrante',
  'Elegante e Sofisticado',
];

const INITIAL_FORM: BriefingFormData = {
  clientName: '',
  clientWhatsapp: '',
  clientEmail: '',
  projectType: PROJECT_TYPES[0],
  businessGoal: '',
  desiredFeatures: ['Botão direto para WhatsApp', 'Otimização para aparecer no Google (SEO)'],
  designStyle: DESIGN_STYLES[0],
  hasDomainOrBrand: false,
  additionalNotes: '',
};

export function BriefingWizard() {
  const [formData, setFormData] = useState<BriefingFormData>(INITIAL_FORM);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [submitted, setSubmitted] = useState<BriefingSubmissionResult | null>(null);

  function toggleFeature(feature: string) {
    setFormData((prev) => {
      const exists = prev.desiredFeatures.includes(feature);
      return {
        ...prev,
        desiredFeatures: exists
          ? prev.desiredFeatures.filter((item) => item !== feature)
          : [...prev.desiredFeatures, feature],
      };
    });
  }

  function handleWhatsappChange(e: React.ChangeEvent<HTMLInputElement>) {
    const masked = maskPhoneBr(e.target.value);
    setFormData((prev) => ({ ...prev, clientWhatsapp: masked }));
  }

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    setError(null);

    const digits = extractPhoneDigits(formData.clientWhatsapp);
    if (digits.length < 10) {
      setError('Digite um número de celular válido com DDD, ex.: (31) 99999-9999.');
      return;
    }

    setLoading(true);

    try {
      const response = await fetch('/api/briefings', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          ...formData,
          clientWhatsapp: digits,
        }),
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        setError(data.message || 'Verifique os campos preenchidos e tente novamente.');
        setLoading(false);
        return;
      }

      const submission = data as BriefingSubmissionResult;
      setSubmitted(submission);

      if (submission.whatsappUrl) {
        const popup = window.open(submission.whatsappUrl, '_blank', 'noopener,noreferrer');
        if (!popup) {
          window.location.href = submission.whatsappUrl;
        }
      }
    } catch {
      setError('Falha de conexão ao enviar o briefing. Tente novamente.');
    } finally {
      setLoading(false);
    }
  }

  function handleReset() {
    setSubmitted(null);
    setError(null);
    setFormData(INITIAL_FORM);
  }

  return (
    <section id="briefing" className="border-t border-slate-200 bg-slate-950 py-20 text-white">
      <div className="mx-auto max-w-4xl px-4 sm:px-6">
        <div className="text-center">
          <span className="inline-block rounded-full bg-blue-500/15 px-3.5 py-1 text-xs font-semibold text-blue-300">
            Solicitar Orçamento
          </span>
          <h2 className="mt-3 text-3xl font-bold tracking-tight sm:text-4xl">
            Conte o que você precisa para o seu site
          </h2>
          <p className="mx-auto mt-3 max-w-2xl text-sm text-slate-300 sm:text-base">
            Preencha este formulário rápido. Ao enviar, os detalhes seguem direto para o WhatsApp do
            FC Studio para iniciarmos o atendimento do seu projeto.
          </p>
        </div>

        <div className="mt-10 rounded-3xl border border-slate-800 bg-slate-900/90 p-6 shadow-2xl sm:p-10">
          {!submitted ? (
            <form onSubmit={handleSubmit} className="space-y-6">
              <div className="grid gap-5 sm:grid-cols-2">
                <div>
                  <label className="block text-sm font-medium text-slate-200">
                    Seu nome ou empresa *
                  </label>
                  <input
                    type="text"
                    required
                    placeholder="Ex.: Mariana Silva / Studio Bella"
                    value={formData.clientName}
                    onChange={(e) => setFormData({ ...formData, clientName: e.target.value })}
                    className="mt-1.5 w-full rounded-xl border border-slate-700 bg-slate-950 px-4 py-3 text-sm text-white placeholder-slate-500 focus:border-blue-500 focus:outline-none"
                  />
                </div>

                <div>
                  <label
                    htmlFor="clientWhatsapp"
                    className="block text-sm font-medium text-slate-200"
                  >
                    Seu WhatsApp (com DDD) *
                  </label>
                  <input
                    id="clientWhatsapp"
                    type="tel"
                    inputMode="numeric"
                    autoComplete="tel"
                    required
                    maxLength={15}
                    placeholder="Ex.: (31) 99999-9999"
                    value={formData.clientWhatsapp}
                    onChange={handleWhatsappChange}
                    className="mt-1.5 w-full rounded-xl border border-slate-700 bg-slate-950 px-4 py-3 text-sm text-white placeholder-slate-500 focus:border-blue-500 focus:outline-none"
                  />
                </div>
              </div>

              <div className="grid gap-5 sm:grid-cols-2">
                <div>
                  <label className="block text-sm font-medium text-slate-200">
                    Qual tipo de site você deseja? *
                  </label>
                  <select
                    value={formData.projectType}
                    onChange={(e) => setFormData({ ...formData, projectType: e.target.value })}
                    className="mt-1.5 w-full rounded-xl border border-slate-700 bg-slate-950 px-4 py-3 text-sm text-white focus:border-blue-500 focus:outline-none"
                  >
                    {PROJECT_TYPES.map((type) => (
                      <option key={type} value={type}>
                        {type}
                      </option>
                    ))}
                  </select>
                </div>

                <div>
                  <label className="block text-sm font-medium text-slate-200">
                    Estilo visual preferido
                  </label>
                  <select
                    value={formData.designStyle}
                    onChange={(e) => setFormData({ ...formData, designStyle: e.target.value })}
                    className="mt-1.5 w-full rounded-xl border border-slate-700 bg-slate-950 px-4 py-3 text-sm text-white focus:border-blue-500 focus:outline-none"
                  >
                    {DESIGN_STYLES.map((style) => (
                      <option key={style} value={style}>
                        {style}
                      </option>
                    ))}
                  </select>
                </div>
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-200">
                  Qual é o principal objetivo desse site para o seu negócio? *
                </label>
                <textarea
                  required
                  rows={3}
                  placeholder="Ex.: Atrair clientes locais para minha clínica, apresentar meus serviços com credibilidade e receber pedidos de orçamento direto no WhatsApp."
                  value={formData.businessGoal}
                  onChange={(e) => setFormData({ ...formData, businessGoal: e.target.value })}
                  className="mt-1.5 w-full rounded-xl border border-slate-700 bg-slate-950 px-4 py-3 text-sm text-white placeholder-slate-500 focus:border-blue-500 focus:outline-none"
                />
              </div>

              <div>
                <span className="block text-sm font-medium text-slate-200">
                  Recursos que você gostaria de incluir
                </span>
                <div className="mt-2.5 grid gap-2.5 sm:grid-cols-2">
                  {FEATURE_OPTIONS.map((feature) => {
                    const checked = formData.desiredFeatures.includes(feature);
                    return (
                      <button
                        type="button"
                        key={feature}
                        onClick={() => toggleFeature(feature)}
                        className={`flex items-center gap-2.5 rounded-xl border px-3.5 py-2.5 text-left text-xs font-medium transition ${
                          checked
                            ? 'border-blue-500 bg-blue-500/15 text-blue-200'
                            : 'border-slate-800 bg-slate-950/70 text-slate-300 hover:border-slate-700'
                        }`}
                      >
                        <span
                          className={`flex h-4 w-4 shrink-0 items-center justify-center rounded border ${
                            checked
                              ? 'border-blue-400 bg-blue-500 text-white'
                              : 'border-slate-600'
                          }`}
                        >
                          {checked ? '✓' : ''}
                        </span>
                        <span>{feature}</span>
                      </button>
                    );
                  })}
                </div>
              </div>

              <div className="flex items-center gap-3 rounded-xl border border-slate-800 bg-slate-950/60 p-4">
                <input
                  id="hasDomain"
                  type="checkbox"
                  checked={formData.hasDomainOrBrand}
                  onChange={(e) =>
                    setFormData({ ...formData, hasDomainOrBrand: e.target.checked })
                  }
                  className="h-4 w-4 rounded border-slate-600 bg-slate-900 text-blue-600"
                />
                <label htmlFor="hasDomain" className="text-sm text-slate-300">
                  Já possuo logotipo, cores da marca ou domínio registrado
                </label>
              </div>

              <div>
                <label className="block text-sm font-medium text-slate-200">
                  Referências ou detalhes extras (opcional)
                </label>
                <textarea
                  rows={2}
                  placeholder="Ex.: Links de sites que você gosta, prazo desejado ou funcionalidades específicas."
                  value={formData.additionalNotes}
                  onChange={(e) => setFormData({ ...formData, additionalNotes: e.target.value })}
                  className="mt-1.5 w-full rounded-xl border border-slate-700 bg-slate-950 px-4 py-3 text-sm text-white placeholder-slate-500 focus:border-blue-500 focus:outline-none"
                />

  return value.replace(/\D/g, '').slice(0, 11);

export function maskPhoneBr(value: string): string {
  const digits = extractPhoneDigits(value);

  if (digits.length === 0) return '';
  if (digits.length <= 2) return `(${digits}`;
  if (digits.length <= 6) return `(${digits.slice(0, 2)}) ${digits.slice(2)}`;
  if (digits.length <= 10) {
    return `(${digits.slice(0, 2)}) ${digits.slice(2, 6)}-${digits.slice(6)}`;
  }

  return `(${digits.slice(0, 2)}) ${digits.slice(2, 7)}-${digits.slice(7, 11)}`;
}

  landing_page: 1100,
  institutional: 2200,
  custom_system_ecommerce: 4200,

const MONTHLY_CARE_BRL: Record<ProjectTier, number> = {

  institutional: 250,

export function mapProjectTypeToTier(projectType: string): ProjectTier {
  const normalized = projectType.toLowerCase();
  if (normalized.includes('landing')) return 'landing_page';
  if (
    normalized.includes('loja') ||
    normalized.includes('e-commerce') ||
    normalized.includes('sistema') ||
    normalized.includes('catálogo')
  ) {
    return 'custom_system_ecommerce';
  }
  return 'institutional';
}

export function calculateProjectEstimate(input: PricingEstimateInput): PricingEstimateOutput {
  const baseDevelopmentBrl = BASE_PRICES_BRL[input.tier] ?? BASE_PRICES_BRL.institutional;
  const safeFeaturesCount = Math.max(0, Math.min(Number(input.featuresCount) || 0, 20));
  const extraFeatures = Math.max(0, safeFeaturesCount - 2);
  const featuresAdjustmentBrl = extraFeatures * 180;

  const domainCostBrl = input.needsCustomDomain
    ? Math.max(0, Number(input.estimatedDomainYearlyBrl) || 45)
    : 0;

  const totalInitialEstimateBrl = baseDevelopmentBrl + featuresAdjustmentBrl + domainCostBrl;
  const suggestedMonthlyCareBrl =
    input.includeMonthlyCare === false
      ? 0
      : MONTHLY_CARE_BRL[input.tier] ?? MONTHLY_CARE_BRL.institutional;

  const notes: string[] = [
    'Infraestrutura inicial em planos gratuitos (Vercel + Turso/Supabase Free) quando compatível com o volume do projeto.',
    'Inclui arquitetura modular, verificação pré-entrega e garantia inicial contra falhas do código entregue.',
  ];

  if (input.needsCustomDomain) {
    notes.push(
      `Inclui estimativa de domínio próprio (${domainCostBrl.toLocaleString('pt-BR', {
        style: 'currency',
        currency: 'BRL',

    );
  } else {
    notes.push('Sem custo de domínio próprio incluído nesta estimativa.');
  }

  return {
    baseDevelopmentBrl,
    featuresAdjustmentBrl,
    domainCostBrl,
    totalInitialEstimateBrl,
    suggestedMonthlyCareBrl,
    notes,
  };
}

  };
}

interface RateLimitEntry {
  count: number;
  resetAt: number;
}

const buckets = new Map<string, RateLimitEntry>();

export function checkRateLimit(
  key: string,
  maxRequests = 30,
  windowMs = 60_000
): { allowed: boolean; retryAfterSec: number } {
  const now = Date.now();

  if (buckets.size > 500) {
    for (const [bucketKey, entry] of buckets.entries()) {
      if (entry.resetAt <= now) {
        buckets.delete(bucketKey);
      }
    }
  }

  const current = buckets.get(key);
  if (!current || current.resetAt <= now) {
    buckets.set(key, { count: 1, resetAt: now + windowMs });
    return { allowed: true, retryAfterSec: 0 };
  }

  if (current.count >= maxRequests) {
    const retryAfterSec = Math.max(1, Math.ceil((current.resetAt - now) / 1000));
    return { allowed: false, retryAfterSec };
  }

  current.count += 1;
  return { allowed: true, retryAfterSec: 0 };
}

export function extractClientIp(request: Request): string {
  const forwardedFor = request.headers.get('x-forwarded-for');
  if (forwardedFor) {
    const first = forwardedFor.split(',')[0]?.trim();
    if (first) return first.slice(0, 80);
  }
  const realIp = request.headers.get('x-real-ip')?.trim();
  return realIp ? realIp.slice(0, 80) : 'local-client';
}

export async function listAdminBriefings(limit = 50): Promise<StoredBriefingRecord[]> {
  const client = getTursoClient();
  if (!client) return [];

  await ensureSchemaAndStatuses();
  const safeLimit = Math.min(Math.max(Number(limit) || 20, 1), 100);

  const result = await client.execute({
    sql: `SELECT
      id,
      client_name,
      client_whatsapp,
      client_email,
      project_type,
      business_goal,
      desired_features,
      design_style,
      has_domain_or_brand,
      additional_notes,
      generated_gemini_prompt,
      status,
      agreed_amount_brl,
      commercial_notes,
      created_at
    FROM project_briefings
    ORDER BY created_at DESC
    LIMIT ?`,
    args: [safeLimit],
  });

  return result.rows.map((row) => {
    let parsedFeatures: string[] = [];
    try {
      parsedFeatures = JSON.parse(String(row.desired_features || '[]'));
    } catch {
      parsedFeatures = [];
    }

    const rawStatus = String(row.status || 'novo');
    const normalizedStatus: BriefingStatus = isValidBriefingStatus(rawStatus)
      ? rawStatus
      : 'novo';

    const rawAmount = row.agreed_amount_brl;
    const numericAmount =
      rawAmount === null || rawAmount === undefined || rawAmount === ''
        ? null
        : Number(rawAmount);

    return {
      id: String(row.id),
      client_name: String(row.client_name),
      client_whatsapp: String(row.client_whatsapp),
      client_email: row.client_email ? String(row.client_email) : null,
      project_type: String(row.project_type),
      business_goal: String(row.business_goal),
      desired_features: Array.isArray(parsedFeatures) ? parsedFeatures : [],
      design_style: String(row.design_style),
      has_domain_or_brand: Number(row.has_domain_or_brand) === 1,
      additional_notes: row.additional_notes ? String(row.additional_notes) : null,
      generated_gemini_prompt: String(row.generated_gemini_prompt),
      status: normalizedStatus,
      agreed_amount_brl:
        numericAmount !== null && Number.isFinite(numericAmount) && numericAmount >= 0
          ? numericAmount
          : null,
      commercial_notes: row.commercial_notes ? String(row.commercial_notes) : null,
      created_at: String(row.created_at),
    };
  });
}

export async function updateBriefingStatus(
  id: string,
  status: BriefingStatus
): Promise<{ ok: boolean; reason?: string }> {
  const client = getTursoClient();
  if (!client) {
    return { ok: false, reason: 'TURSO_ENV_NOT_CONFIGURED' };
  }

  if (!(DB_ALLOWED_STATUSES as readonly string[]).includes(status)) {
    return { ok: false, reason: 'INVALID_STATUS' };
  }

  try {
    await ensureSchemaAndStatuses();
    const res = await client.execute({
      sql: 'UPDATE project_briefings SET status = ? WHERE id = ?',
      args: [status, id],
    });

    if ((res.rowsAffected ?? 0) === 0) {
      return { ok: false, reason: 'NOT_FOUND' };
    }

    return { ok: true };
  } catch {
    schemaChecked = false;
    try {
      await ensureSchemaAndStatuses();
      const retry = await client.execute({
        sql: 'UPDATE project_briefings SET status = ? WHERE id = ?',
        args: [status, id],
      });
      return (retry.rowsAffected ?? 0) > 0
        ? { ok: true }
        : { ok: false, reason: 'NOT_FOUND' };
    } catch {
      return { ok: false, reason: 'TURSO_UPDATE_ERROR' };
    }
  }
}

export async function auditAndSyncProjectBriefingsSchema(): Promise<TursoSchemaAuditResult> {
  const client = getTursoClient();
  if (!client) {
    return {
      configured: false,
      columnsOk: false,
      statusConstraintOk: false,
      indexOk: false,
      totalRows: 0,
      missingColumns: EXPECTED_COLUMNS,
    };
  }

  await client.execute({ sql: CREATE_BRIEFINGS_SQL, args: [] });

  const infoResult = await client.execute({
    sql: "SELECT sql FROM sqlite_master WHERE type = 'table' AND name = 'project_briefings'",
    args: [],
  });
  const tableSql = String(infoResult.rows[0]?.sql || '');
  const statusConstraintOk = tableSql.includes("'feito'") && tableSql.includes("'em_producao'");

  if (!statusConstraintOk && tableSql.length > 0) {
    await client.batch(
      [
        "PRAGMA foreign_keys = OFF",
        CREATE_TMP_BRIEFINGS_SQL,
        COPY_BRIEFINGS_SQL,
        "DROP TABLE project_briefings",
        "ALTER TABLE project_briefings_sync_tmp RENAME TO project_briefings",
        "PRAGMA foreign_keys = ON",
      ],
      'write'
    );
  }

  const colsRes = await client.execute({
    sql: "PRAGMA table_info(project_briefings)",
    args: [],
  });
  const colSet = new Set(colsRes.rows.map((r) => String(r.name || '').toLowerCase()));

  if (!colSet.has('agreed_amount_brl')) {
    await client.execute({
      sql: "ALTER TABLE project_briefings ADD COLUMN agreed_amount_brl REAL",
      args: [],
    });
    colSet.add('agreed_amount_brl');
  }

  if (!colSet.has('commercial_notes')) {
    await client.execute({
      sql: "ALTER TABLE project_briefings ADD COLUMN commercial_notes TEXT",
      args: [],
    });
    colSet.add('commercial_notes');
  }

  await client.execute({ sql: CREATE_INDEX_SQL, args: [] });

  const idxRes = await client.execute({
    sql: "SELECT name FROM sqlite_master WHERE type = 'index' AND name = 'idx_project_briefings_status_created'",
    args: [],
  });

  const countRes = await client.execute({
    sql: "SELECT COUNT(*) AS total FROM project_briefings",
    args: [],
  });

  const missingColumns = EXPECTED_COLUMNS.filter((col) => !colSet.has(col));
  const finalInfo = await client.execute({
    sql: "SELECT sql FROM sqlite_master WHERE type = 'table' AND name = 'project_briefings'",
    args: [],
  });
  const finalSql = String(finalInfo.rows[0]?.sql || '');

  return {
    configured: true,
    columnsOk: missingColumns.length === 0,
    statusConstraintOk: finalSql.includes("'feito'") && finalSql.includes("'em_producao'"),
    indexOk: idxRes.rows.length > 0,
    totalRows: Number(countRes.rows[0]?.total || 0),
    missingColumns,
  };
}
